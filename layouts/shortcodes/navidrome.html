{{- .Page.Store.Set "hasNavidromePlayer" true -}} {{- $url := strings.TrimSpace (.Get "url") -}} {{- $startRaw := strings.TrimSpace (.Get "start" | default "0") -}} {{- $allowedPrefix := .Site.Params.navidromeSharePrefix | default "https://music.tommasofrancescon.it/share/" -}} {{- if not (hasPrefix $url $allowedPrefix) -}} {{- errorf "navidrome shortcode: 'url' deve iniziare con %q (%s)" $allowedPrefix .Position -}} {{- end -}} {{- $startAt := 0 -}} {{- if findRE `^\d+$` $startRaw -}} {{- $startAt = int $startRaw -}} {{- else if findRE `^\d+:[0-5]?\d$` $startRaw -}} {{- $parts := split $startRaw ":" -}} {{- $startAt = add (mul (int (index $parts 0)) 60) (int (index $parts 1)) -}} {{- else if findRE `^\d+:[0-5]\d:[0-5]\d$` $startRaw -}} {{- $parts := split $startRaw ":" -}} {{- $startAt = add (mul (int (index $parts 0)) 3600) (mul (int (index $parts 1)) 60) (int (index $parts 2)) -}} {{- else -}} {{- errorf "navidrome shortcode: 'start' deve essere espresso in secondi, m:ss o h:mm:ss (%s)" .Position -}} {{- end -}} {{- $remote := try (resources.GetRemote $url) -}} {{- if $remote.Err -}} {{- warnf "Impossibile leggere la condivisione Navidrome %q: %s" $url $remote.Err -}}

{{ i18n "navidromeUnavailable" }} {{ i18n "navidromeOpenShare" }}.

{{- else if not $remote.Value -}} {{- warnf "La condivisione Navidrome %q non ha restituito contenuti" $url -}}

{{ i18n "navidromeUnavailable" }} {{ i18n "navidromeOpenShare" }}.

{{- else -}} {{- $html := $remote.Value.Content -}} {{- $matches := findRESubmatch `(?s)window\.__SHARE_INFO__\s*=\s*(.*?)\s*` $html 1 -}} {{- if not $matches -}} {{- errorf "navidrome shortcode: metadati non trovati in %q; verifica che sia una condivisione Navidrome pubblica (%s)" $url .Position -}} {{- else -}} {{- $share := transform.Unmarshal (index (index $matches 0) 1) -}} {{- if eq (printf "%T" $share) "string" -}} {{- $share = transform.Unmarshal $share -}} {{- end -}} {{- $tracks := slice -}} {{- if reflect.IsMap $share -}} {{- with index $share "tracks" -}} {{- if reflect.IsSlice . -}} {{- $tracks = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq (len $tracks) 0 -}} {{- errorf "navidrome shortcode: la condivisione %q non contiene brani o usa un formato non supportato (%s)" $url .Position -}} {{- else if ne (len $tracks) 1 -}} {{- errorf "navidrome shortcode: la condivisione %q deve contenere un solo brano (%s)" $url .Position -}} {{- else -}} {{- $track := index $tracks 0 -}} {{- $duration := float $track.duration -}} {{- if le $duration 0.0 -}} {{- errorf "navidrome shortcode: durata del brano non valida nella condivisione %q (%s)" $url .Position -}} {{- else if ge (float $startAt) $duration -}} {{- errorf "navidrome shortcode: 'start' (%s) deve essere inferiore alla durata del brano (%s secondi) (%s)" $startRaw $duration .Position -}} {{- else -}} {{- $shareRoot := replaceRE `/[^/]+/?$` "" $url -}} {{- $streamURL := printf "%s/s/%s" $shareRoot $track.id -}} {{- $coverURL := printf "%s/img/%s?size=360" $shareRoot $track.id -}} {{- $playerID := printf "navidrome-player-%d" .Ordinal -}} {{- end }} {{- end -}} {{- end -}} {{- end -}}
